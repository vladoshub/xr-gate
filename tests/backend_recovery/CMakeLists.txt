cmake_minimum_required(VERSION 3.20)
project(xr_backend_recovery_tests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(TBB REQUIRED)
get_filename_component(XR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(backend_recovery_test backend_recovery_test.cpp)
target_compile_features(backend_recovery_test PRIVATE cxx_std_17)
target_include_directories(backend_recovery_test PRIVATE
  "${XR_ROOT}/shared/include" "${XR_ROOT}/backends/basalt_vio/include")
target_link_libraries(backend_recovery_test PRIVATE Threads::Threads nlohmann_json::nlohmann_json TBB::tbb)
target_compile_options(backend_recovery_test PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
add_test(NAME backend_recovery COMMAND backend_recovery_test)
set_tests_properties(backend_recovery PROPERTIES TIMEOUT 15)
add_library(fake_mercury SHARED fake_mercury.cpp)
target_compile_features(fake_mercury PRIVATE cxx_std_17)
target_include_directories(fake_mercury PRIVATE "${XR_ROOT}/shared/include"
  "${XR_ROOT}/backends/mercury_hand_tracking/include")
target_link_libraries(fake_mercury PRIVATE nlohmann_json::nlohmann_json ${CMAKE_DL_LIBS})
