cmake_minimum_required(VERSION 3.16)
project(xr_gate_hand_skeleton LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This example currently supports Linux X11/GLX (including an XWayland session).")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPENXR REQUIRED IMPORTED_TARGET openxr)
find_package(SDL2 REQUIRED)
find_package(OpenGL REQUIRED)
find_package(GLEW REQUIRED)
find_package(X11 REQUIRED)
add_executable(xr_gate_hand_skeleton src/main.cpp)
target_compile_features(xr_gate_hand_skeleton PRIVATE cxx_std_17)
target_compile_options(xr_gate_hand_skeleton PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
target_link_libraries(xr_gate_hand_skeleton PRIVATE PkgConfig::OPENXR SDL2::SDL2 OpenGL::GL GLEW::GLEW X11::X11)
install(TARGETS xr_gate_hand_skeleton RUNTIME DESTINATION bin)
